cmake_minimum_required(VERSION 3.21)

project(
	hello_pong
	LANGUAGES    C CXX ASM
	VERSION      1.0.0
	DESCRIPTION  "hello_pong"
	HOMEPAGE_URL "http://lameguy64.net/?page=psn00bsdk"
)

file(GLOB _sources *.c)
file(GLOB _lib_sources libs/*.c)

psn00bsdk_add_executable(${PROJECT_NAME} GPREL ${_sources} ${_lib_sources})

# Auto-detect TIM assets, link them, and generate assets/assets.h
file(GLOB TIM_ASSETS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" CONFIGURE_DEPENDS "assets/*.tim")
set(ASSET_HEADER_CONTENT "// Auto-generated by CMake. Do not edit manually.\n#ifndef ASSETS_H\n#define ASSETS_H\n\n#include <sys/types.h>\n\n")

foreach(TIM_FILE ${TIM_ASSETS})
	get_filename_component(ASSET_NAME "${TIM_FILE}" NAME_WLE)
	psn00bsdk_target_incbin(${PROJECT_NAME} PRIVATE tim_${ASSET_NAME} ${TIM_FILE})
	string(APPEND ASSET_HEADER_CONTENT "extern u_long tim_${ASSET_NAME}[];\n")
endforeach()

string(APPEND ASSET_HEADER_CONTENT "\n#endif // ASSETS_H\n")
file(WRITE "${CMAKE_CURRENT_SOURCE_DIR}/assets/assets.h" "${ASSET_HEADER_CONTENT}")

psn00bsdk_add_cd_image(
	iso         # Target name
	${PROJECT_NAME} # Output file name (= <name>.bin + <name>.cue)
	iso.xml     # Path to config file
	DEPENDS ${PROJECT_NAME} system.cnf
)
